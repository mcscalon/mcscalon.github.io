rank_by_year <- tuesdata$rank_by_year

lowest.rank <- rank_by_year %>% 
  group_by(country) %>% 
  summarise_if(is.numeric, mean) %>% 
  slice_max(order_by = rank, n = 5)

lowest <- rank_by_year %>% 
  filter(country %in% lowest.rank$country)

highest.rank <- rank_by_year %>% 
  group_by(country) %>% 
  summarise_if(is.numeric, mean) %>% 
  slice_min(order_by = rank, n = 5)

highest <- rank_by_year %>%
